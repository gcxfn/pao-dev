import { revisionLabel } from "./memory-labels";
import { useEffect, useRef, useState } from "react";
import type {
  PaoMemoryDeleteConfirmResult,
  PaoMemoryDeletePreviewResult,
  PaoMemoryDeletionOperation,
  PaoMemoryListItem,
  PaoMemoryUsageState,
  PaoReadDesktopMemoryDetailResult,
  PaoSharedSourcePolicy,
} from "@pao/sdk-driver";
import type { MemoryMutationResult } from "../../../contracts/memory";
import type { PiDesktopApi } from "../../../contracts/ipc";
import { useI18n } from "../../i18n/react";
import { formatMemoryTime } from "./memory-list";
import {
  additionalMemoryDeletionTargets,
  buildMemoryCorrectionOp,
  canConfirmMemoryDeletion,
  createMemoryCorrectionDraft,
  describeMemoryDeletionPendingKind,
  describeMemoryDeletionSessionAvailability,
  describeMemoryDeletionSourceAction,
  describeMemoryDeletionSourceResolution,
  hasCurrentRevisionConflict,
  isMemoryRevisionConflict,
  memoryDeleteSource,
  memoryOperationSource,
  selectableMergeCandidates,
  validateMemoryMerge,
  type MemoryCorrectionDraft,
} from "./memory-management-model";

type EditorMode = "none" | "correct" | "merge";
type BusyOperation =
  | "correct"
  | "merge"
  | "usage"
  | "deletePreview"
  | "deleteConfirm"
  | "deleteQuery"
  | "deleteResume";

interface CorrectionConflict {
  readonly message: string;
  readonly latest?: PaoReadDesktopMemoryDetailResult;
  readonly loading: boolean;
}

export interface MemoryManagementProps {
  readonly api: PiDesktopApi;
  readonly workspaceId: string;
  readonly detail: PaoReadDesktopMemoryDetailResult;
  readonly mergeCandidates: readonly PaoMemoryListItem[];
  readonly allowMerge: boolean;
  readonly onMutationComplete: (result: MemoryMutationResult, memoryId: string) => void;
  readonly onDeletionOperation: (operation: PaoMemoryDeletionOperation) => void;
}

export function useMemoryManagement({
  api,
  workspaceId,
  detail,
  mergeCandidates,
  allowMerge,
  onMutationComplete,
  onDeletionOperation,
}: MemoryManagementProps) {
  const { language, t } = useI18n();
  const copy = (zh: string, en: string) => (language === "zh" ? zh : en);
  const [mode, setMode] = useState<EditorMode>("none");
  const [busy, setBusy] = useState<BusyOperation>();
  const [error, setError] = useState<string>();
  const [notice, setNotice] = useState<string>();
  const [correctionDraft, setCorrectionDraft] = useState<MemoryCorrectionDraft>();
  const [correctionConflict, setCorrectionConflict] = useState<CorrectionConflict>();
  const [conflictAcknowledged, setConflictAcknowledged] = useState(false);
  const [mergeId, setMergeId] = useState<string>();
  const [keepId, setKeepId] = useState(detail.memory.id);
  const [mergeRationale, setMergeRationale] = useState("");
  const [deletePreview, setDeletePreview] = useState<PaoMemoryDeletePreviewResult>();
  const [deletePreviewPolicy, setDeletePreviewPolicy] = useState<PaoSharedSourcePolicy>();
  const [deletionResult, setDeletionResult] = useState<PaoMemoryDeleteConfirmResult>();
  const [sharedSourcePolicy, setSharedSourcePolicy] = useState<PaoSharedSourcePolicy>("retain");
  const [recursiveTargetAcknowledged, setRecursiveTargetAcknowledged] = useState(false);
  const correctionTriggerRef = useRef<HTMLButtonElement>(null);
  const mergeTriggerRef = useRef<HTMLButtonElement>(null);
  const correctionFormRef = useRef<HTMLFormElement>(null);
  const mergeFormRef = useRef<HTMLFormElement>(null);

  useEffect(() => {
    setMode("none");
    setBusy(undefined);
    setError(undefined);
    setNotice(undefined);
    setCorrectionDraft(undefined);
    setCorrectionConflict(undefined);
    setConflictAcknowledged(false);
    setMergeId(undefined);
    setKeepId(detail.memory.id);
    setMergeRationale("");
    setDeletePreview(undefined);
    setDeletePreviewPolicy(undefined);
    setDeletionResult(undefined);
    setSharedSourcePolicy("retain");
    setRecursiveTargetAcknowledged(false);
  }, [detail.memory.id, workspaceId]);

  useEffect(() => {
    if (mode === "none") return;
    const form = mode === "correct" ? correctionFormRef.current : mergeFormRef.current;
    form?.querySelector<HTMLElement>("textarea, input, select")?.focus();
  }, [mode]);

  const operationSource = memoryOperationSource();
  const candidateItems = selectableMergeCandidates(detail.memory.id, mergeCandidates);
  const selectedMergeItem = candidateItems.find((item) => item.memoryId === mergeId);
  const mergeUnavailable = Boolean(busy) || !allowMerge || candidateItems.length === 0;

  const beginCorrection = () => {
    setMode("correct");
    setCorrectionDraft(createMemoryCorrectionDraft(detail));
    setCorrectionConflict(undefined);
    setConflictAcknowledged(false);
    setError(undefined);
    setNotice(undefined);
  };

  const restoreEditorFocus = (editor: Exclude<EditorMode, "none">) => {
    requestAnimationFrame(() => {
      (editor === "correct" ? correctionTriggerRef : mergeTriggerRef).current?.focus();
    });
  };

  const closeEditor = () => {
    if (busy || mode === "none") return;
    const closedEditor = mode;
    setMode("none");
    setError(undefined);
    restoreEditorFocus(closedEditor);
  };

  const submitCorrection = async () => {
    if (!correctionDraft || busy) return;
    if (!correctionDraft.conclusion.trim()) {
      setError(copy("纠正结论不能为空。", "The corrected conclusion cannot be empty."));
      return;
    }
    setBusy("correct");
    setError(undefined);
    setNotice(undefined);
    try {
      const result = await api.mutateMemory({
        workspaceId,
        op: buildMemoryCorrectionOp(correctionDraft),
      });
      if (isMemoryRevisionConflict(result)) {
        setCorrectionConflict({
          message:
            "op" in result && result.ok === false
              ? result.error.message
              : copy("记忆已被其他操作更新。", "The memory changed in another operation."),
          loading: true,
        });
        setConflictAcknowledged(false);
        const latest = await api.getMemoryDetail({
          workspaceId,
          id: correctionDraft.memoryId,
        });
        setCorrectionConflict((current) =>
          current
            ? {
                ...current,
                ...(latest.ok ? { latest } : {}),
                loading: false,
              }
            : current,
        );
        if (!latest.ok) {
          setError(
            copy(
              "已检测到版本冲突，但最新版本读取失败：",
              "Revision conflict detected; loading the latest revision failed: ",
            ) + latest.error.message,
          );
        }
        return;
      }
      if (!result.ok) {
        setError(result.error.message);
        return;
      }
      if (!("op" in result) || result.op !== "correct") {
        setError(copy("服务未返回纠正确认。", "The service did not confirm the correction."));
        return;
      }
      setNotice(
        copy(
          `已保存为版本 ${revisionLabel(result.revision)}。`,
          `Saved as revision ${revisionLabel(result.revision)}.`,
        ),
      );
      setMode("none");
      setCorrectionConflict(undefined);
      restoreEditorFocus("correct");
      onMutationComplete(result, detail.memory.id);
    } catch (reason) {
      setError(reason instanceof Error ? reason.message : String(reason));
    } finally {
      setBusy(undefined);
    }
  };

  const acknowledgeLatestCorrection = () => {
    const latest = correctionConflict?.latest;
    if (!latest || !correctionDraft) return;
    setCorrectionDraft({
      ...correctionDraft,
      expectedRevision: latest.currentRevision,
    });
    setConflictAcknowledged(true);
    setError(undefined);
  };

  const submitMerge = async () => {
    if (!selectedMergeItem || busy) return;
    const validation = validateMemoryMerge(
      { keepId, mergeId: selectedMergeItem.memoryId },
      mergeRationale,
    );
    if (validation) {
      setError(
        copy(
          "请选择两条不同记忆、明确保留项并填写合并理由。",
          "Select two different memories, an explicit item to keep, and a rationale.",
        ),
      );
      return;
    }
    setBusy("merge");
    setError(undefined);
    setNotice(undefined);
    try {
      const result = await api.mutateMemory({
        workspaceId,
        op: {
          op: "merge",
          keepId,
          keepExpectedRevision:
            keepId === detail.memory.id ? detail.currentRevision : selectedMergeItem.revision,
          mergeId: selectedMergeItem.memoryId,
          mergeExpectedRevision:
            selectedMergeItem.memoryId === detail.memory.id
              ? detail.currentRevision
              : selectedMergeItem.revision,
          rationale: mergeRationale.trim(),
          source: operationSource,
        },
      });
      if (!result.ok) {
        setError(
          isMemoryRevisionConflict(result)
            ? copy(
                "合并项已变化，当前选择与理由已保留；请刷新后重新确认。",
                "A merge item changed. Your selection and rationale were retained; refresh and confirm again.",
              )
            : result.error.message,
        );
        return;
      }
      if (!("op" in result) || result.op !== "merge") {
        setError(copy("服务未返回合并确认。", "The service did not confirm the merge."));
        return;
      }
      setNotice(
        copy(
          `已合并到保留项的版本 ${revisionLabel(result.keepRevision)}。`,
          `Merged into the retained revision ${revisionLabel(result.keepRevision)}.`,
        ),
      );
      setMode("none");
      restoreEditorFocus("merge");
      onMutationComplete(result, keepId);
    } catch (reason) {
      setError(reason instanceof Error ? reason.message : String(reason));
    } finally {
      setBusy(undefined);
    }
  };

  const setUsage = async (target: PaoMemoryUsageState) => {
    if (busy) return;
    setBusy("usage");
    setError(undefined);
    setNotice(undefined);
    try {
      const result = await api.mutateMemory({
        workspaceId,
        op: {
          op: "setUsage",
          id: detail.memory.id,
          expectedRevision: detail.currentRevision,
          target,
          source: operationSource,
        },
      });
      if (!result.ok) {
        setError(result.error.message);
        return;
      }
      if (!("op" in result) || result.op !== "setUsage") {
        setError(copy("服务未返回状态更新确认。", "The service did not confirm the usage change."));
        return;
      }
      setNotice(
        target === "stopped"
          ? copy(
              `已停用，当前版本 ${revisionLabel(result.revision)}。`,
              `Stopped at revision ${revisionLabel(result.revision)}.`,
            )
          : copy(
              `已恢复使用，当前版本 ${revisionLabel(result.revision)}。`,
              `Restored at revision ${revisionLabel(result.revision)}.`,
            ),
      );
      onMutationComplete(result, detail.memory.id);
    } catch (reason) {
      setError(reason instanceof Error ? reason.message : String(reason));
    } finally {
      setBusy(undefined);
    }
  };

  const previewDeletion = async (policy: PaoSharedSourcePolicy = sharedSourcePolicy) => {
    if (busy) return;
    setBusy("deletePreview");
    setError(undefined);
    setNotice(undefined);
    setDeletionResult(undefined);
    setSharedSourcePolicy(policy);
    setDeletePreviewPolicy(undefined);
    setRecursiveTargetAcknowledged(false);
    try {
      const result = await api.mutateMemory({
        workspaceId,
        op: {
          op: "deletePreview",
          memoryId: detail.memory.id,
          sharedSourcePolicy: policy,
        },
      });
      if (!result.ok) {
        setError(result.error.message);
        return;
      }
      if (!("op" in result) || result.op !== "deletePreview") {
        setError(copy("服务未返回删除预览。", "The service did not return a deletion preview."));
        return;
      }
      setDeletePreview(result);
      setDeletePreviewPolicy(policy);
      setDeletionResult(undefined);
    } catch (reason) {
      setError(reason instanceof Error ? reason.message : String(reason));
    } finally {
      setBusy(undefined);
    }
  };

  const confirmDeletion = async () => {
    if (!deletePreview || busy) return;
    const memoryIds = deletePreview.targets.map((target) => target.memoryId);
    const revisions = Object.fromEntries(
      deletePreview.targets.map((target) => [target.memoryId, target.currentRevision]),
    );
    setBusy("deleteConfirm");
    setError(undefined);
    try {
      const result = await api.mutateMemory({
        workspaceId,
        op: {
          op: "deleteConfirm",
          memoryIds,
          revisions,
          source: memoryDeleteSource(),
          sharedSourcePolicy,
        },
      });
      if (!result.ok) {
        setError(result.error.message);
        return;
      }
      if (!("op" in result) || result.op !== "deleteConfirm") {
        setError(copy("服务未返回删除操作状态。", "The service did not return a deletion status."));
        return;
      }
      setDeletionResult(result);
      onDeletionOperation(result);
      if (result.status === "complete") {
        setNotice(copy("删除已完成。", "Deletion completed."));
        onMutationComplete(result, detail.memory.id);
      }
    } catch (reason) {
      setError(reason instanceof Error ? reason.message : String(reason));
    } finally {
      setBusy(undefined);
    }
  };

  const queryDeletion = async () => {
    if (!deletionResult || busy) return;
    setBusy("deleteQuery");
    setError(undefined);
    try {
      const result = await api.getMemoryDeletionOperation({
        workspaceId,
        operationId: deletionResult.operationId,
      });
      if (!result.ok) {
        setError(result.error.message);
        return;
      }
      setDeletionResult(result);
      onDeletionOperation(result);
      if (result.status === "complete") {
        setNotice(copy("删除已完成。", "Deletion completed."));
        onMutationComplete({ op: "deleteConfirm", ...result }, detail.memory.id);
      }
    } catch (reason) {
      setError(reason instanceof Error ? reason.message : String(reason));
    } finally {
      setBusy(undefined);
    }
  };

  const resumeDeletion = async () => {
    if (!deletionResult || busy) return;
    setBusy("deleteResume");
    setError(undefined);
    try {
      const result = await api.resumeMemoryDeletion({
        workspaceId,
        operationId: deletionResult.operationId,
      });
      if (!result.ok) {
        setError(result.error.message);
        return;
      }
      setDeletionResult(result);
      onDeletionOperation(result);
      if (result.status === "complete") {
        setNotice(copy("删除已完成。", "Deletion completed."));
        onMutationComplete({ op: "deleteConfirm", ...result }, detail.memory.id);
      }
    } catch (reason) {
      setError(reason instanceof Error ? reason.message : String(reason));
    } finally {
      setBusy(undefined);
    }
  };

  const additionalTargets = deletePreview
    ? additionalMemoryDeletionTargets(deletePreview, detail.memory.id)
    : [];
  const recursiveConfirmationRequired =
    sharedSourcePolicy === "include-referencing-memories" && additionalTargets.length > 0;
  const deletionCanBeConfirmed = canConfirmMemoryDeletion({
    preview: deletePreview,
    previewedPolicy: deletePreviewPolicy,
    policy: sharedSourcePolicy,
    currentMemoryId: detail.memory.id,
    recursiveTargetAcknowledged,
    busy: Boolean(busy),
    hasOperationResult: Boolean(deletionResult),
  });

  return {
    detail,
    allowMerge,
    language,
    t,
    copy,
    mode,
    setMode,
    busy,
    error,
    setError,
    notice,
    setNotice,
    correctionDraft,
    setCorrectionDraft,
    correctionConflict,
    conflictAcknowledged,
    mergeId,
    setMergeId,
    keepId,
    setKeepId,
    mergeRationale,
    setMergeRationale,
    deletePreview,
    setDeletePreview,
    setDeletePreviewPolicy,
    deletionResult,
    sharedSourcePolicy,
    recursiveTargetAcknowledged,
    setRecursiveTargetAcknowledged,
    correctionTriggerRef,
    mergeTriggerRef,
    correctionFormRef,
    mergeFormRef,
    candidateItems,
    selectedMergeItem,
    mergeUnavailable,
    beginCorrection,
    closeEditor,
    submitCorrection,
    acknowledgeLatestCorrection,
    submitMerge,
    setUsage,
    previewDeletion,
    confirmDeletion,
    queryDeletion,
    resumeDeletion,
    additionalTargets,
    recursiveConfirmationRequired,
    deletionCanBeConfirmed,
  };
}

export type MemoryManagementState = ReturnType<typeof useMemoryManagement>;

export function MemoryManagement({
  state,
  placement,
}: {
  readonly state: MemoryManagementState;
  readonly placement: "header-actions" | "forms";
}) {
  const {
    detail,
    allowMerge,
    language,
    t,
    copy,
    mode,
    setMode,
    busy,
    error,
    setError,
    notice,
    setNotice,
    correctionDraft,
    setCorrectionDraft,
    correctionConflict,
    conflictAcknowledged,
    mergeId,
    setMergeId,
    keepId,
    setKeepId,
    mergeRationale,
    setMergeRationale,
    deletePreview,
    setDeletePreview,
    setDeletePreviewPolicy,
    deletionResult,
    sharedSourcePolicy,
    recursiveTargetAcknowledged,
    setRecursiveTargetAcknowledged,
    correctionTriggerRef,
    mergeTriggerRef,
    correctionFormRef,
    mergeFormRef,
    candidateItems,
    selectedMergeItem,
    mergeUnavailable,
    beginCorrection,
    closeEditor,
    submitCorrection,
    acknowledgeLatestCorrection,
    submitMerge,
    setUsage,
    previewDeletion,
    confirmDeletion,
    queryDeletion,
    resumeDeletion,
    additionalTargets,
    recursiveConfirmationRequired,
    deletionCanBeConfirmed,
  } = state;
  if (detail.isHistorical || detail.readOnly) return null;
  if (placement === "header-actions")
    return (
      <div className="memory-management__actions">
        <button
          ref={correctionTriggerRef}
          type="button"
          className="button"
          data-testid="memory-action-correct"
          disabled={Boolean(busy)}
          onClick={beginCorrection}
        >
          {copy("纠正结论", "Correct")}
        </button>
        <button
          ref={mergeTriggerRef}
          type="button"
          className="button"
          data-testid="memory-action-merge"
          aria-disabled={mergeUnavailable}
          title={
            allowMerge
              ? candidateItems.length === 0
                ? copy(
                    "当前没有可合并的其他记忆。",
                    "No other current memories are available to merge.",
                  )
                : undefined
              : copy("请在仅当前版的列表视图执行合并。", "Merge from current-only list mode.")
          }
          onClick={() => {
            if (mergeUnavailable) return;
            setMode("merge");
            setKeepId(detail.memory.id);
            setMergeId(undefined);
            setMergeRationale("");
            setError(undefined);
            setNotice(undefined);
          }}
        >
          {copy("合并两条", "Merge two")}
        </button>
        <button
          type="button"
          className="button"
          data-testid="memory-action-usage"
          disabled={Boolean(busy)}
          onClick={() => {
            setUsage(detail.memory.usageState === "active" ? "stopped" : "active").catch(() => {});
          }}
        >
          {busy === "usage"
            ? t("Loading…")
            : detail.memory.usageState === "active"
              ? copy("停用", "Stop")
              : copy("恢复使用", "Restore")}
        </button>
        <button
          type="button"
          className="button button--danger"
          data-testid="memory-action-delete"
          disabled={Boolean(busy)}
          onClick={() => {
            previewDeletion().catch(() => {});
          }}
        >
          {busy === "deletePreview"
            ? copy("正在预览…", "Preparing preview…")
            : copy("彻底删除", "Delete permanently")}
        </button>
      </div>
    );
  return (
    <section className="memory-management" data-testid="memory-management">
      {notice ? (
        <div
          className="memory-inline-state memory-inline-state--success"
          role="status"
          data-testid="memory-management-notice"
        >
          {notice}
        </div>
      ) : null}
      {error ? (
        <div
          className="memory-inline-state memory-inline-state--error"
          role="alert"
          data-testid="memory-management-error"
        >
          {error}
        </div>
      ) : null}

      {mode === "correct" && correctionDraft ? (
        <form
          ref={correctionFormRef}
          className="memory-editor"
          data-testid="memory-correction-form"
          onKeyDown={(event) => {
            if (event.key === "Escape") {
              event.preventDefault();
              closeEditor();
            }
          }}
          onSubmit={(event) => {
            event.preventDefault();
            submitCorrection().catch(() => {});
          }}
        >
          <div className="memory-editor__head">
            <div>
              <h3>{copy("纠正当前记忆", "Correct this memory")}</h3>
              <p>
                {copy("打开时版本", "Opened at revision")}{" "}
                {revisionLabel(correctionDraft.expectedRevision)}
              </p>
            </div>
            <button type="button" className="button" disabled={Boolean(busy)} onClick={closeEditor}>
              {copy("关闭", "Close")}
            </button>
          </div>
          <label className="memory-editor__field">
            <span>{copy("纠正结论", "Corrected conclusion")}</span>
            <textarea
              aria-label={copy("纠正结论", "Corrected conclusion")}
              rows={3}
              value={correctionDraft.conclusion}
              onChange={(event) =>
                setCorrectionDraft({ ...correctionDraft, conclusion: event.currentTarget.value })
              }
            />
          </label>
          <label className="memory-editor__field">
            <span>{copy("适用条件", "Conditions")}</span>
            <textarea
              aria-label={copy("适用条件", "Conditions")}
              rows={2}
              value={correctionDraft.conditions}
              onChange={(event) =>
                setCorrectionDraft({ ...correctionDraft, conditions: event.currentTarget.value })
              }
            />
          </label>
          <label className="memory-editor__field">
            <span>{copy("纠正理由", "Rationale")}</span>
            <textarea
              aria-label={copy("纠正理由", "Rationale")}
              rows={2}
              value={correctionDraft.rationale}
              onChange={(event) =>
                setCorrectionDraft({ ...correctionDraft, rationale: event.currentTarget.value })
              }
            />
          </label>
          <div className="memory-editor__row">
            <label className="memory-editor__field">
              <span>{copy("确认信息", "Confirmation")}</span>
              <select
                aria-label={copy("确认信息", "Confirmation")}
                value={correctionDraft.confirmation}
                onChange={(event) =>
                  setCorrectionDraft({
                    ...correctionDraft,
                    confirmation: event.currentTarget
                      .value as MemoryCorrectionDraft["confirmation"],
                  })
                }
              >
                <option value="confirmed">{t("Confirmed")}</option>
                <option value="tentative">{t("Tentative")}</option>
                <option value="rejected">{t("Rejected")}</option>
              </select>
            </label>
            <div className="memory-editor__evidence">
              <span>{copy("证据类别", "Evidence kind")}</span>
              <strong>{t("User statement")}</strong>
              <small>{copy("由用户纠正语义确定", "Set by this user-correction action")}</small>
            </div>
          </div>

          {correctionConflict ? (
            <div className="memory-conflict" role="alert" data-testid="memory-correction-conflict">
              <strong>
                {copy("检测到版本冲突；草稿已保留", "Revision conflict; your draft was retained")}
              </strong>
              <p>{correctionConflict.message}</p>
              {correctionConflict.loading ? (
                <p>{copy("正在读取最新版本…", "Loading the latest revision…")}</p>
              ) : correctionConflict.latest ? (
                <div className="memory-conflict__latest" data-testid="memory-correction-latest">
                  <span>
                    {copy("最新版本", "Latest revision")} r
                    {correctionConflict.latest.currentRevision}
                  </span>
                  <strong>{correctionConflict.latest.memory.content.conclusion}</strong>
                  <p>
                    {copy("最新条件：", "Latest conditions: ")}
                    {correctionConflict.latest.memory.content.conditions ||
                      copy("未记录", "Not recorded")}
                  </p>
                  <p>
                    {copy("最新理由：", "Latest rationale: ")}
                    {correctionConflict.latest.memory.content.rationale ||
                      copy("未记录", "Not recorded")}
                  </p>
                  <time>{formatMemoryTime(correctionConflict.latest.memory.updatedAt)}</time>
                  <button
                    type="button"
                    className="button"
                    data-testid="memory-conflict-continue"
                    disabled={conflictAcknowledged || Boolean(busy)}
                    onClick={acknowledgeLatestCorrection}
                  >
                    {conflictAcknowledged
                      ? copy("已采用最新版作为保存基准", "Using latest as save baseline")
                      : copy("基于最新版本继续", "Continue from latest revision")}
                  </button>
                </div>
              ) : null}
            </div>
          ) : null}

          <div className="memory-editor__footer">
            <button type="submit" className="button button--primary" disabled={Boolean(busy)}>
              {busy === "correct"
                ? copy("保存中…", "Saving…")

              {deletionResult.status !== "complete" ? (

                    {busy === "deleteQuery"
                      ? copy("查询中…", "Checking…")
                      : copy("查询当前状态", "Check status")}
                  </button>
                  <button
                    type="button"

                      resumeDeletion().catch(() => {});
                    }}

                    {busy === "deleteResume"
                      ? copy("恢复中…", "Resuming…")
                      : copy("恢复删除作业", "Resume deletion")}
                  </button>
                </div>
              ) : null}
            </div>
          ) : (
            <div className="memory-delete-preview__footer">
              <button
                type="button"
                className="button button--danger"

                disabled={!deletionCanBeConfirmed}

                  : copy("确认彻底删除", "Confirm permanent deletion")}
              </button>

        </section>
      ) : null}
    </section>
  );
}

export function MemoryDeletionRecovery({
  api,
  workspaceId,
  operationIds,
  onOperationUpdate,
  onCompleted,
}: {
  readonly api: PiDesktopApi;
  readonly workspaceId: string;
  readonly operationIds: readonly string[];
  readonly onOperationUpdate: (operation: PaoMemoryDeletionOperation) => void;
  readonly onCompleted: (operation: PaoMemoryDeletionOperation) => void;
}) {
  const { language } = useI18n();
  const copy = (zh: string, en: string) => (language === "zh" ? zh : en);
  const [operations, setOperations] = useState<
    Readonly<Record<string, PaoMemoryDeletionOperation>>
  >({});
  const [errors, setErrors] = useState<Readonly<Record<string, string>>>({});
  const [busyId, setBusyId] = useState<string>();
  const operationKey = operationIds.join("\0");

  useEffect(() => {
    let active = true;
    setOperations({});
    setErrors({});
    Promise.all(
      operationIds.map(async (operationId) => {
        const result = await api.getMemoryDeletionOperation({ workspaceId, operationId });
        return { operationId, result };
      }),
    )
      .then((entries) => {
        if (!active) return;
        const nextOperations: Record<string, PaoMemoryDeletionOperation> = {};
        const nextErrors: Record<string, string> = {};
        for (const { operationId, result } of entries) {
          if (result.ok) {
            nextOperations[operationId] = result;
            onOperationUpdate(result);
            if (result.status === "complete") onCompleted(result);
          } else {
            nextErrors[operationId] = result.error.message;
          }
        }
        setOperations(nextOperations);
        setErrors(nextErrors);
      })
      .catch(() => {});
    return () => {
      active = false;
    };
  }, [api, onCompleted, onOperationUpdate, operationIds, operationKey, workspaceId]);

  if (operationIds.length === 0) return null;

  const resume = async (operationId: string) => {
    if (busyId) return;
    setBusyId(operationId);
    setErrors((current) => {
      const next = { ...current };
      delete next[operationId];
      return next;
    });
    try {
      const result = await api.resumeMemoryDeletion({ workspaceId, operationId });
      if (!result.ok) {
        setErrors((current) => ({
          ...current,
          [operationId]: result.error.message,
        }));
        return;
      }
      setOperations((current) => ({ ...current, [operationId]: result }));
      onOperationUpdate(result);
      if (result.status === "complete") onCompleted(result);
    } catch (reason) {
      setErrors((current) => ({
        ...current,
        [operationId]: reason instanceof Error ? reason.message : String(reason),
      }));
    } finally {
      setBusyId(undefined);
    }
  };

  return (
    <section
      className="memory-delete-recovery"
      data-testid="memory-delete-recovery"
      aria-label={copy("删除作业恢复", "Deletion recovery")}
    >
      <header>
        <div>
          <strong>{copy("尚未完成的删除", "Unfinished deletions")}</strong>
          <span>
            {copy(
              "这里保留仍需继续处理的删除；确认完成后会自动从本区移除。",
              "Deletions that still need attention remain here and leave automatically once confirmed complete.",
            )}
          </span>
        </div>
      </header>
      <ul>
        {operationIds.map((operationId) => {
          const operation = operations[operationId];
          const error = errors[operationId];
          return (
            <li
              key={operationId}
              className={`memory-delete-recovery__item memory-delete-recovery__item--${operation?.status ?? "loading"}`}
              data-testid="memory-delete-recovery-item"
              data-operation-id={operationId}
              data-status={operation?.status ?? "loading"}
            >
              <div>
                <strong>
                  {operation?.status === "pending"
                    ? copy("等待继续处理", "Waiting to continue")
                    : operation?.status === "failed"
                      ? copy("删除失败", "Failed")
                      : copy("正在查询", "Checking")}
                </strong>
              </div>
              {operation?.failureReason ? (
                <details className="memory-technical-details">
                  <summary>{copy("失败详情", "Failure details")}</summary>
                  <p>{operation.failureReason}</p>
                </details>
              ) : null}
              {operation?.pendingObjects.length ? (
                <ul>
                  {operation.pendingObjects.map((pending, index) => (
                    <li
                      key={`${pending.kind}:${pending.sessionId ?? ""}:${pending.sourceId ?? ""}:${index}`}
                    >
                      <span>{describeMemoryDeletionPendingKind(pending.kind, language)}</span>
                      <details className="memory-technical-details">
                        <summary>{copy("技术信息", "Technical details")}</summary>
                        <dl>
                          <dt>{copy("对象类型", "Object type")}</dt>
                          <dd>{describeMemoryDeletionPendingKind(pending.kind, language)}</dd>
                          {pending.sessionId ? (
                            <>
                              <dt>{copy("会话 ID", "Session ID")}</dt>
                              <dd>{pending.sessionId}</dd>
                            </>
                          ) : null}
                          {pending.sourceId ? (
                            <>
                              <dt>{copy("来源 ID", "Source ID")}</dt>
                              <dd>{pending.sourceId}</dd>
                            </>
                          ) : null}
                          {pending.entryId ? (
                            <>
                              <dt>{copy("条目 ID", "Entry ID")}</dt>
                              <dd>{pending.entryId}</dd>
                            </>
                          ) : null}
                          <dt>{copy("原因", "Reason")}</dt>
                          <dd>{pending.reason}</dd>
                        </dl>
                      </details>
                    </li>
                  ))}
                </ul>
              ) : null}
              <details className="memory-technical-details">
                <summary>{copy("任务详情", "Operation details")}</summary>
                <p>
                  {copy("任务编号", "Operation ID")}: <code>{operationId}</code>
                </p>
              </details>
              {operation ? (
                <small>
                  {copy("更新时间", "Updated")} {formatMemoryTime(operation.updatedAt)}
                </small>
              ) : null}
              {error ? (
                <p className="memory-inline-state memory-inline-state--error" role="alert">
                  {error}
                </p>
              ) : null}
              <button
                type="button"
                className="button button--primary"
                disabled={Boolean(busyId) || !operation}
                onClick={() => {
                  resume(operationId).catch(() => {});
                }}
              >
                {busyId === operationId
                  ? copy("继续处理中…", "Continuing…")
                  : copy("继续删除", "Continue deletion")}
              </button>
            </li>
          );
        })}
      </ul>
    </section>
  );
}
