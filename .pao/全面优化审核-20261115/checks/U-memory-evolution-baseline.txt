/**
 * 版本目录种子：版本目录 = <home>/evolution/versions/<dirSafe>/<versionId>/，
 * dirSafe = resourceId 的 ":" 换成 "_"。非 v-<sha12> 形态的 versionId 走
 * 目录非空降级判据，写一个占位文件即可让 restore 判据 b 通过。
 */
async function seedVersionDirs(
  home: string,
  resourceId: string,
  versionIds: readonly string[],
): Promise<void> {
  const dirSafe = resourceId.replace(/:/g, "_");
  for (const versionId of versionIds) {
    const dir = join(home, "evolution", "versions", dirSafe, versionId);
    await mkdir(dir, { recursive: true });
    await writeFile(join(dir, "SKILL.md"), `# ${versionId}\n`, "utf8");
  }
}

    for (const label of ["发布", "生效", "绑定", "激活", "Publish", "Activate", "Bind"]) {
      await expect(detail.getByRole("button", { name: label })).toHaveCount(0);
    }
    // 候选状态不能把 passed 显示成已生效。
    await expect(candidate).not.toContainText("已生效");
    await expect(candidate).not.toContainText("active");
  } finally {
    await app.close();
  }
});

// ── S9-E：生效与观察 ─────────────────────────────────────────────────────
// 公共前置：建用户数据/项目/PAO_HOME，灌进化种子 + 版本目录，启动桌面并
// 选中 evolution 列表里的 beta 问题行（resourceId=skill:beta）。

async function launchSeededEvolutionPage() {
  const userDataDir = await makeUserDataDir("memory-evolution-s9-user-data-");
  const project = await makeWorkspace("memory-evolution-s9-project");
  const home = await mkdtemp(join(tmpdir(), "pao-evolution-s9-home-"));
  await writeFile(join(userDataDir, "ui-state.json"), JSON.stringify({ language: "zh" }));

  const app = await launchDesktop(userDataDir, {
    initialWorkspaces: [project],
    envOverrides: {
      PAO_HOME: home,
      PAO_DISABLE_BUILTIN_EXTENSION: "1",
      PI_OFFLINE: "1",
      PI_NO_LOCAL_LLM: "1",
    },
    scrubProviderEnv: true,
  });
  const page = await app.firstWindow();
  await page.setViewportSize({ width: 1440, height: 900 });
  await page.evaluate(async () => {
    const state = await window.piApp.getState();
    const target = state.workspaces.find(
      (workspace) => workspace.name === "memory-evolution-s9-project",
    );
    if (!target) throw new Error("memory-evolution-s9-project was not registered");
    if (state.selectedWorkspaceId !== target.id) {
      await window.piApp.selectWorkspace(target.id, { stayInView: true });
    }
  });

  const dist = join(__dirname, "../../node_modules/@pao/coding-agent/dist/pao");
  const seedPath = join(home, "seed-evolution.mjs");
  await writeFile(seedPath, seedScript);
  const sessionId = (await getDesktopState(page)).selectedSessionId ?? "seed-session";
  await execFileAsync(process.execPath, [seedPath, dist, project, home, sessionId], {
    windowsHide: true,
    env: { ...process.env, PAO_HOME: home },
  });
  // 版本目录：v-beta-0/1/2 都需要目录存在（非指纹形态走非空判据）。
  await seedVersionDirs(home, "skill:beta", ["v-beta-0", "v-beta-1", "v-beta-2"]);

  await page.getByTestId("sidebar-memory").click();
  await expect(page.getByTestId("memory-page")).toBeVisible();
  await page.getByLabel("类型").selectOption("evolution");
  await expect(page.getByTestId("evolution-problem-list")).toBeVisible();
  const betaRow = page
    .getByTestId("evolution-problem-item")
    .filter({ hasText: "beta 技能调用方式问题" });
  await betaRow.click();
  const detail = page.getByTestId("evolution-detail");
  await expect(detail).toBeVisible();
  const panel = detail.getByTestId("evolution-resource");
  await expect(panel).toBeVisible();
  return { app, page, detail, panel, home, project };
}

test("S9-E 版本冲突时保留当前值", async ({}) => {
  test.setTimeout(180_000);
  const { app, page, panel } = await launchSeededEvolutionPage();
  try {
    // 模拟过期 expectedEffectiveVersion：先把开关切走再切回不改 effectiveVersion，
    // 真正的冲突来自 restoreVersion 的 expectedEffectiveVersion 与当前指向不符。
    // 用 IPC 直接发一个错误 expected 的 restore，断言 conflict 且指向不变。
    const outcome = await page.evaluate(async () => {
      const state = await window.piApp.getState();
      const ws = state.workspaces.find(
        (workspace) => workspace.name === "memory-evolution-s9-project",
      );
      if (!ws) throw new Error("workspace missing");
      const next = await window.piApp!.mutateMemoryEvolution({
        workspaceId: ws.id,
        action: {
          action: "restoreVersion",
          resourceId: "skill:beta",
          targetVersionId: "v-beta-1",
          expectedEffectiveVersion: "v-stale-not-current",
        },
      });
      if (!next.ok || next.action !== "restoreVersion") {
        return { ok: false as const };
      }
      return {
        ok: true as const,
        restored: next.restored,
        status: next.status,
        currentEffectiveVersion: next.currentEffectiveVersion,
      };
    });
    expect(outcome.ok).toBe(true);
    if (outcome.ok) {
      expect(outcome.restored).toBe(false);
      expect(outcome.status).toBe("conflict");
      expect(outcome.currentEffectiveVersion).toBe("v-beta-2");
    }
    // 当前值保留：重新拉资源确认 effectiveVersion 仍是 v-beta-2。
    await expect(panel.getByTestId("evolution-effective-version")).toContainText("v-beta-2");
  } finally {
    await app.close();
  }
});
