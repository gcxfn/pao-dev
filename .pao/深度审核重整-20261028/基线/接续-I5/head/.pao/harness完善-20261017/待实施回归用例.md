# 待实施回归用例

这些是本轮新增的五个用例，已从活动测试文件撤回，供后续实施者使用。四项已反向运行并复现原缺陷，确认等待取消项尚未运行。需要原测试文件的 setup/call/mockPi 等助手。

```typescript
it("宿主项目不能被工具参数改写，拒绝前不创建另一个项目的计划", async () => {
		const f = await setup();
		const otherProject = tempDir();
		await expect(
			call(f.tools, "pao_plan_create", {
				project_root: otherProject,
				request_id: "cross-project",
				display_name: "越界创建",
				steps: [{ title: "写作", dept: "写作部", requirement: "写初稿" }],
			}),
		).rejects.toThrow(/项目.*不一致|项目.*不符/);
		expect(existsSync(join(otherProject, ".pao"))).toBe(false);
	});

	it("其他主控不能修改、撤销、启动或写意见到原主控的计划", async () => {
		const f = await setup();
		const { 计划ID } = await 建计划(f.tools);
		const { tools, seen } = withConfirmation({ ...f.options, resolveOwnerSessionId: () => "sess-other" }, f.common);
		const before = readTemporaryPlan(f.项目根, 计划ID);
		for (const action of ["modify", "undo"]) {
			await expect(
				call(tools, "pao_plan_update", {
					plan_id: 计划ID, request_id: `foreign-${action}`, expected_version: 1, action,
					...(action === "modify" ? { display_name: "不该写入" } : {}),
				}),
			).rejects.toThrow(/主控|owner|归属/);
		}
		await expect(call(tools, "pao_plan_start", { plan_id: 计划ID, definition_version: 1, start_request_id: "foreign-start" })).rejects.toThrow(/主控|owner|归属/);
		await expect(call(tools, "pao_plan_feedback", { plan_id: 计划ID, action: "record", step_id: "s1", text: "不该写入" })).rejects.toThrow(/主控|owner|归属/);
		expect(seen).toHaveLength(0);
		expect(readTemporaryPlan(f.项目根, 计划ID)).toEqual(before);
	});

	it("确认等待期间取消立即返回，迟到的同意不创建实例", async () => {
		const f = await setup();
		const { 计划ID } = await 建计划(f.tools);
		const { pi, tools } = mockPi();
		let entered!: () => void;
		let agree!: (value: import("../../src/pao/workflow/plan.ts").PlanConfirmation) => void;
		const waiting = new Promise<void>((resolve) => { entered = resolve; });
		registerPlanTools(pi, { ...f.options, requestConfirmation: () => { entered(); return new Promise((resolve) => { agree = resolve; }); } }, f.common);
		const controller = new AbortController();
		const pending = tools.get("pao_plan_start")!.execute("cancel", { plan_id: 计划ID, definition_version: 1, start_request_id: "cancel-wait" } as never, controller.signal);
		await waiting;
		controller.abort(new Error("用户取消启动"));
		await expect(pending).rejects.toThrow("用户取消启动");
		agree({ 计划ID, 定义版本: 1, 启动请求ID: "cancel-wait", 来源: "宿主确认回调", 时刻: "2026-10-17 12:00" });
		await Promise.resolve();
		expect(existsSync(planExecDir(f.项目根, 计划ID, "cancel-wait"))).toBe(false);
	});

	it("确认期间原主控发生切换，确认回来后拒绝启动", async () => {
		const f = await setup();
		const { 计划ID } = await 建计划(f.tools);
		const { pi, tools } = mockPi();
		let owner = "sess-1";
		registerPlanTools(pi, {
			...f.options, resolveOwnerSessionId: () => owner,
			requestConfirmation: (request) => {
				owner = "sess-other";
				return { ...request, 来源: "宿主确认回调", 时刻: "2026-10-17 12:00" };
			},
		}, f.common);
		await expect(call(tools, "pao_plan_start", { plan_id: 计划ID, definition_version: 1, start_request_id: "owner-change" })).rejects.toThrow(/主控|owner|归属/);
		expect(existsSync(planExecDir(f.项目根, 计划ID, "owner-change"))).toBe(false);
	});

	it("控制正本受理后即冻结；等待后台推进不能留出可编辑窗口", async () => {
		const f = await setup();
		const { 计划ID } = await 建计划(f.tools);
		let entered!: () => void;
		let release!: () => void;
		const enteredBackground = new Promise<void>((resolve) => { entered = resolve; });
		const pendingBackground = new Promise<void>((resolve) => { release = resolve; });
		const common: WorkflowCommon = { queueReview: () => {}, background: {
			accept: async () => { entered(); await pendingBackground; return "离线后台受理"; },
			waitForAutoChain: async () => {},
		} };
		const { tools } = withConfirmation(f.options, common);
		const pending = call(tools, "pao_plan_start", { plan_id: 计划ID, definition_version: 1, start_request_id: "freeze-window" });
		await enteredBackground;
		try {
			await expect(call(f.tools, "pao_plan_update", { plan_id: 计划ID, request_id: "during-start", expected_version: 1, action: "modify", display_name: "未获准的新版本" })).rejects.toThrow(/已启动|冻结/);
			expect(readTemporaryPlan(f.项目根, 计划ID).定义版本).toBe(1);
		} finally {
			release();
			await pending;
		}
	});
```
